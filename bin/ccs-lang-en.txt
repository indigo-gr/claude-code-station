# --- --help
help=ccs v{version} - Claude Code Station (fzf-powered launcher)
help+=
help+=Usage:
help+=  ccs                 Mixed list: NEW repos + RESUME sessions
help+=  ccs .               Sessions whose cwd is (under) current dir
help+=  ccs --new           Only NEW repo entries
help+=  ccs --resume        Only past sessions
help+=  ccs --refresh       Force pre-scan before showing list
help+=  ccs --no-scan       Skip implicit pre-scan (use stale DB)
help+=  ccs --lang=ja|en    Language of ccs's own text for this run (wins over
help+=                      CCS_LANG and defaults.language in repos.yml)
help+=  ccs init            Create ~/.config/ccs/repos.yml from the template
help+=  ccs init --auto-discover [--depth N]
help+=                      Scan $HOME for git repos (default depth 5) and pick
help+=                      which to add to repos.yml via fzf multi-select
help+=  ccs [options]       Unknown options pass through to launched command
help+=
help+=For scripts and tools (no fzf, no questions, no network; --json prints one
help+=JSON answer on stdout, and the exit code says what happened):
help+=  ccs doctor [--json] Does ccs work here? Exit 0 works, 1 does not,
help+=                      2 bad arguments, 3 could not tell
help+=  ccs list --json [--with-paths]
help+=                      repos.yml entries after ccs's checks (paths only
help+=                      with --with-paths)
help+=  ccs sessions --json [--project NAME] [--limit N] [--with-titles] [--no-scan]
help+=                      Past conversations per project, newest first
help+=                      (titles only with --with-titles)
help+=  ccs repo add PATH [--name NAME] [--description TEXT] [--json]
help+=  ccs repo remove NAME [--json]
help+=                      Add a folder under $HOME to repos.yml, or take an
help+=                      entry out (the folder is never touched)
help+=  ccs config get [--json]
help+=  ccs config set [--repo NAME] KEY VALUE [--json]
help+=                      KEY is defaults.ui, defaults.language,
help+=                      defaults.terminal, defaults.update_check, or
help+=                      label / description of the entry named by --repo
help+=  ccs open (--repo NAME | --session UUID) [--here | --new-window]
help+=                      Start Claude Code without the list: in this terminal
help+=                      (--here, the default), or in a new window of
help+=                      defaults.terminal (--new-window, macOS)
help+=  Exit codes: 0 ok, 1 failed, 2 bad arguments, 3 could not tell,
help+=  4 missing prerequisite, 5 not found, 6 refused, 7 partly done,
help+=  130 interrupted. doctor, list, sessions, repo, config and open are
help+=  commands only as the first argument; `ccs --new doctor` still passes
help+=  doctor on. --json output never changes with the language.
help+=
help+=Controls:
help+=  Enter               Launch (or resume) selected entry; does nothing on a
help+=                      group header (an entry with launch: false)
help+=  Ctrl-Y              Copy the effective shell command to clipboard
help+=  Ctrl-I              Copy session UUID (resume) or repo path (new/group)
help+=  Ctrl-R              Refresh DB and reload list
help+=  Ctrl-D              Delete session (resume rows only)
help+=  Esc / Ctrl-C        Cancel
help+=
help+=Environment:
help+=  CCS_CMD             Fallback launch command for repos with no explicit
help+=                      command. Per-repo command in repos.yml always wins.
help+=                      Example: export CCS_CMD="opr claude"
help+=  CCR_CMD             Deprecated — still honored with a warning if CCS_CMD unset
help+=  CCS_LANG            ja | en — language of ccs's own text (--lang= wins;
help+=                      then defaults.language, then LC_ALL / LC_MESSAGES / LANG)
help+=  CCS_NO_UPDATE_CHECK Any non-empty value stops the update check and its notice
help+=                      (no network request, the cache is not read); so does
help+=                      defaults.update_check: false in repos.yml
help+=  CCS_UPDATE_REGISTRY Registry to ask instead of https://registry.npmjs.org
help+=
help+=Examples:
help+=  ccs                                     # Full launcher
help+=  ccs .                                   # Resume current-dir session
help+=  ccs --new                               # Pick a repo to launch fresh
help+=  ccs --refresh                           # Force DB rebuild
help+=  ccs --dangerously-skip-permissions      # Pass-through flag to claude
help+=
help+=Config: ~/.config/ccs/repos.yml
help+=Cache:  ~/.cache/ccs/state.db, ~/.cache/ccs/update-check
